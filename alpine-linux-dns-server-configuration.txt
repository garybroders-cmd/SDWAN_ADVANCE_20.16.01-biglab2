>>>> Install the packages below on the Linux Alpine EVE-NG QCOW2 image <<<<

apk update
apk add bind bind-tools

>>>> First, configure the Linux interface eth0 <<<<<
>>>> vi /etc/network/interfaces <<<<<

auto lo
iface lo inet loopback

auto eth0
iface eth0 inet static
  address 39.1.255.222
  netmask 255.255.255.0
  gateway 39.1.255.254

>>>> Reboot the box or service <<<<<

service hostname restart

>>>> create dns resover <<<<<
>>>> vi /etc/resolv.conf <<<<<

search cloud2.local
nameserver 8.8.8.8
nameserver 1.1.1.1
  
>>>> make this into a DNS server forwarder and lookup DNS server <<<<<
>>>> vi /etc/bind/named.conf <<<<<<

options {
directory "/var/bind";
  recursion yes;
  allow-recursion { any; };
  forwarders {
    8.8.8.8;
    1.1.1.1;
  };
forward only;

listen-on { any; };
listen-on-v6 { any; };
rrset-order {
order cyclic;
};
};

zone "cloud2.local" {
  type master;
  file "/etc/bind/db.cloud2.local";
  };

>>>> check the file is correct before restarting <<<<<

named-checkconf
named-checkzone cloud2.local /etc/bind/db.cloud2.local
